// ── Guard against Vite-internal WebSocket frame errors ──────────
// When Vite runs in middlewareMode, its bundled ws instance can emit
// uncaught errors for malformed HMR close frames (Node 18+).
// These are non-fatal and should not crash the server.
process.on('uncaughtException', (err: any) => {
  if (
    err?.code === 'WS_ERR_INVALID_CLOSE_CODE' ||
    err?.message?.includes('Invalid WebSocket frame') ||
    err?.message?.includes('WebSocket')
  ) {
    console.debug('[Server] Non-fatal WebSocket frame error suppressed:', err.message);
    return; // swallow and continue
  }
  // Re-throw anything genuinely unexpected
  console.error('[Server] Uncaught exception:', err);
  process.exit(1);
});

// Real-time AI Image Generation Endpoint for Photos & Realistic Visuals
app.post('/api/generate-image', async (req, res) => {
  const { prompt: userPrompt, topic } = req.body || {};
  const imagePrompt =
    userPrompt ||
    `A high-resolution, photorealistic, scientific educational photo depicting ${topic || 'science subject'}, sharp focus, authentic natural lighting, realistic textures, macro/telephoto lens, no text overlays, cinematic clarity`;

  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) {
    return res.json({
      success: true,
      imageUrl: `https://picsum.photos/seed/${encodeURIComponent(String(topic || 'education').toLowerCase())}/1280/720`,
      caption: `Visual Representation of ${topic || 'Topic'}`,
      promptUsed: imagePrompt,
    });
  }

  try {
    const ai = new GoogleGenAI({
      apiKey,
      httpOptions: {
        headers: { 'User-Agent': 'aistudio-build' },
      },
    });

    // Try nano banana image generation with candidate models
    const imageCandidateModels = ['gemini-3.1-flash-image', 'gemini-3.1-flash-lite-image'];
    let foundBase64: string | null = null;

    for (const imgModel of imageCandidateModels) {
      try {
        const imgResponse = await ai.models.generateContent({
          model: imgModel,
          contents: {
            parts: [{ text: imagePrompt }],
          },
          config: {
            imageConfig: {
              aspectRatio: '16:9',
            },
          },
        });

        const parts = imgResponse.candidates?.[0]?.content?.parts || [];
        for (const part of parts) {
          if (part.inlineData?.data) {
            foundBase64 = part.inlineData.data;
            break;
          }
        }
        if (foundBase64) break;
      } catch (imgErr: any) {
        console.warn(`[API /api/generate-image] Model ${imgModel} returned error, trying next candidate...`, imgErr?.message);
      }
    }

    if (foundBase64) {
      return res.json({
        success: true,
        imageUrl: `data:image/png;base64,${foundBase64}`,
        caption: `AI Generated Real-Time Photographic Study: ${topic || 'Subject'}`,
        promptUsed: imagePrompt,
      });
    }

    // If no inlineData returned, provide high-quality fallback
    return res.json({
      success: true,
      imageUrl: `https://picsum.photos/seed/${encodeURIComponent(String(topic || 'education').toLowerCase())}/1280/720`,
      caption: `Visual Study: ${topic || 'Topic'}`,
      promptUsed: imagePrompt,
    });
  } catch (err: any) {
    console.error('[API /api/generate-image] Image generation failed, using fallback:', err?.message);
    return res.json({
      success: true,
      imageUrl: `https://picsum.photos/seed/${encodeURIComponent(String(topic || 'education').toLowerCase())}/1280/720`,
      caption: `Visual Study: ${topic || 'Topic'}`,
      promptUsed: imagePrompt,
    });
  }

const upload = multer({
  dest: UPLOAD_DIR,
  limits: { fileSize: 500 * 1024 * 1024, files: 10 },
  fileFilter: (_req, file, cb) => {
    const ok = file.mimetype === 'application/pdf' || /\.pdf$/i.test(file.originalname);
    if (ok) cb(null, true);
    else cb(new Error(`"${file.originalname}" is not a PDF`));
  },
});

// POST /api/curriculum/upload — one or more PDFs → background job.
// Accepts the new 'pdfs' field (multiple) and the old single 'pdf' field.
app.post('/api/curriculum/upload',
  upload.fields([{ name: 'pdfs', maxCount: 10 }, { name: 'pdf', maxCount: 1 }]),
  (req: any, res: any) => {
    const files: any[] = [...(req.files?.pdfs || []), ...(req.files?.pdf || [])];
    const cleanup = () => files.forEach(f => fs.promises.unlink(f.path).catch(() => {}));
    const apiKey = process.env.GEMINI_API_KEY;
    if (!apiKey) { cleanup(); return res.status(500).json({ error: 'GEMINI_API_KEY not set on the server' }); }
    if (!files.length) return res.status(400).json({ error: 'No PDF received' });
    const subjectLabel = String(req.body.subjectLabel || '').trim();
    const grade = String(req.body.grade || '').trim();
    // Derived from the subject name, so Book 2A and Book 2B land in the same subject.
    const subjectId = String(req.body.subjectId || subjectLabel).toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
    if (!subjectLabel || !grade || !subjectId) { cleanup(); return res.status(400).json({ error: 'Subject name and grade are required' }); }

    const job = startIngestJob({
      apiKey, subjectId, subjectLabel, grade,
      files: files.map(f => ({ path: f.path, originalName: f.originalname })),
    });
    res.json({ success: true, jobId: job.id, subjectId });
  });

// ─── Adaptive session endpoints ─────────────────────────────────
app.post('/api/session/start', (req, res) => {
  const { studentId, name, grade, subjectId } = req.body;
  if (!studentId || !subjectId)
    return (res as any).status(400).json({ error: 'studentId and subjectId required' });
  const learner = getOrCreateLearner(studentId, name || 'Student', grade || 'Secondary 2 (Grade 8)');
  const curriculum = subjectId === 'pythagoras' ? PYTHAGORAS_CURRICULUM : getCurriculum(subjectId);
  if (!curriculum) return (res as any).status(404).json({ error: 'Curriculum not found' });
  ensureSubject(studentId, subjectId, curriculum.label, curriculum.grade, curriculum.source);

  const masteredIds = Object.values(learner.subjects[subjectId]?.conceptStates || {})
    .filter(cs => cs.masteryScore >= 75).map(cs => cs.conceptId);
  const nextConcept = nextUnmasteredConcept(curriculum, masteredIds) || curriculum.concepts[0];
  ensureConceptState(studentId, subjectId, nextConcept.id, nextConcept.label);

  const sessionId = `session_${Date.now()}_${studentId}`;
  const session: AdaptiveSessionState = {
    sessionId, studentId, subjectId,
    currentConceptId: nextConcept.id,
    currentStrategy: 'direct_explanation',
    sessionStarted: Date.now(),
    interactionCount: 0,
    recentAttempts: [],
  };
  startSession(session);
  res.json({
    sessionId, learner, currentConcept: nextConcept,
    conceptState: getConceptState(studentId, subjectId, nextConcept.id),
    strategy: 'direct_explanation',
  });
});

app.post('/api/session/:sessionId/assess', async (req, res) => {
  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) return (res as any).status(500).json({ error: 'No API key' });
  const session = getSession(req.params.sessionId);
  if (!session) return (res as any).status(404).json({ error: 'Session not found' });

  const { questionSummary, studentAnswer, correctAnswer, selectedOptionIndex, correctOptionIndex } = req.body;
  const isCorrect = selectedOptionIndex === correctOptionIndex;

  const curriculum = session.subjectId === 'pythagoras' ? PYTHAGORAS_CURRICULUM : getCurriculum(session.subjectId);
  const concept = curriculum?.concepts.find(c => c.id === session.currentConceptId);
  if (!concept) return (res as any).status(404).json({ error: 'Concept not found' });
  const conceptState = getConceptState(session.studentId, session.subjectId, concept.id);
  if (!conceptState) return (res as any).status(404).json({ error: 'Concept state not found' });

  const assessment = await assessUnderstanding({
    concept, questionAsked: questionSummary, studentAnswer,
    correctAnswer, isCorrect, selectedOptionIndex, conceptState, apiKey,
  });

  const attempt = {
    timestamp: Date.now(), questionSummary, conceptTag: concept.id,
    selectedOption: selectedOptionIndex, correctOption: correctOptionIndex,
    isCorrect, understandingDepth: assessment.understandingDepth,
    misconceptionDetected: assessment.misconceptionDescription,
    strategyUsed: session.currentStrategy as TeachingStrategy,
    teachingNote: assessment.teachingNote,
  };
  recordAttempt(session.studentId, session.subjectId, concept.id, attempt, assessment);

  const nextStrategy = selectNextStrategy(conceptState, assessment);
  let nextConceptId = session.currentConceptId;
  let advancedToConcept = null;

  if (['advance','praise_and_continue'].includes(assessment.recommendedAction)) {
    const updated = getConceptState(session.studentId, session.subjectId, concept.id);
    if ((updated?.masteryScore || 0) >= 75) {
      const masteredIds = Object.values(
        getLearner(session.studentId)?.subjects?.[session.subjectId]?.conceptStates || {}
      ).filter(cs => cs.masteryScore >= 75).map(cs => cs.conceptId);
      const next = curriculum ? nextUnmasteredConcept(curriculum, masteredIds) : undefined;
      if (next && next.id !== session.currentConceptId) {
        nextConceptId = next.id; advancedToConcept = next;
        ensureConceptState(session.studentId, session.subjectId, next.id, next.label);
      }
    }
  }

  updateSession(session.sessionId, {
    currentConceptId: nextConceptId, currentStrategy: nextStrategy,
    interactionCount: session.interactionCount + 1,
    recentAttempts: [...session.recentAttempts.slice(-4), attempt],
    pendingStrategySwitch: assessment.recommendedAction === 'switch_strategy' ? nextStrategy : undefined,
    switchReason: assessment.recommendedAction === 'switch_strategy' ? assessment.teachingNote : undefined,
  });

  res.json({
    assessment, nextStrategy, advancedToConcept,
    updatedConceptState: getConceptState(session.studentId, session.subjectId, nextConceptId),
    learner: getLearner(session.studentId),
    session: getSession(session.sessionId),
  });
});

app.post('/api/session/:sessionId/end', (req, res) => {
  const session = getSession(req.params.sessionId);
  if (!session) { res.json({ ok: true }); return; }
  const durationMins = Math.round((Date.now() - session.sessionStarted) / 60000);
  incrementSessionCount(session.studentId, session.subjectId, durationMins);
  endSession(session.sessionId);
  res.json({ ok: true, durationMinutes: durationMins });
});

app.get('/api/learner-context/:studentId/:subjectId', (req, res) => {
  const { studentId, subjectId } = req.params;
  const learner = getLearner(studentId);
  if (!learner) return (res as any).status(404).json({ error: 'Learner not found' });
  const subject = learner.subjects[subjectId];
  if (!subject) return res.json({ context: 'No prior learning history for this subject.' });

  const conceptSummaries = Object.values(subject.conceptStates)
    .sort((a, b) => b.lastVisited - a.lastVisited).slice(0, 10)
    .map(cs => {
      const lines = [`- ${cs.label}: mastery ${cs.masteryScore}/100 (${cs.masteryLevel})`];
      if (cs.confirmedMisconceptions.length)
        lines.push(`  CONFIRMED MISCONCEPTIONS: ${cs.confirmedMisconceptions.join('; ')}`);
      if (cs.effectiveStrategies.length)
        lines.push(`  EFFECTIVE STRATEGIES: ${cs.effectiveStrategies.join(', ')}`);
      if (cs.ineffectiveStrategies.length)
        lines.push(`  DID NOT HELP: ${cs.ineffectiveStrategies.join(', ')}`);
      return lines.join('\n');
    }).join('\n');

  const context = `LEARNER: ${learner.name} (${learner.grade})\nSessions: ${subject.sessionCount}, Minutes: ${subject.totalMinutes}\n\nCONCEPT MASTERY:\n${conceptSummaries || 'None yet'}\n\nINSIGHTS: ${learner.globalInsights.slice(-3).join(' | ') || 'None yet'}`;
  res.json({ context });
});

// ─── Voice session log ────────────────────────────────────────
// One file per session in logs/. Timings, tool calls, closes and what was said,
// so a "the tutor went quiet" report can be diagnosed from evidence.
// Local development only; the folder is git-ignored.
function openVoiceLog(topic: string) {
  const dir = path.join(process.cwd(), 'logs');
  try { fs.mkdirSync(dir, { recursive: true }); } catch (_) {}
  const stamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19);
  const file = path.join(dir, `voice-${stamp}.log`);
  const t0 = Date.now();
  // Buffered stream: a synchronous write per event would stall the same
  // event loop that relays audio between the browser and Gemini.
  const stream = fs.createWriteStream(file, { flags: 'a' });
  stream.on('error', () => {});
  const write = (line: string) => { stream.write(line + '\n'); };
  write(`# voice session ${new Date().toISOString()} topic="${topic}"`);
  return (event: string, data?: Record<string, any>) => {
    const t = ((Date.now() - t0) / 1000).toFixed(2).padStart(8);
    const body = data ? ' ' + JSON.stringify(data).slice(0, 400) : '';
    write(`${t}s  ${event}${body}`);
    if (!/^(tutor_said|child_said)$/.test(event)) console.log(`[voice] ${event}${body.slice(0, 160)}`);
  };
}

// WebSocket Server for Live API audio and events
const wss = new WebSocketServer({ noServer: true });
wss.on('error', (err) => {
  console.error('[WSS] Server error:', err.message);
});

// Fallback WebSocket server for Vite HMR / browser client keepalive
const fallbackWss = new WebSocketServer({
  noServer: true,
  handleProtocols: (protocols: Set<string>) => {
    if (protocols.has('vite-hmr')) return 'vite-hmr';
    const first = Array.from(protocols)[0];
    return first || false;
  },
});
fallbackWss.on('connection', (ws) => {
  try {
    ws.send(JSON.stringify({ type: 'connected' }));
  } catch (_) {}
  ws.on('message', () => {});
  // Prevent unhandled 'error' events from crashing the process
  // (Vite HMR client can send malformed close frames locally)
  ws.on('error', (err) => {
    console.debug('[FallbackWS] Non-fatal WebSocket error (Vite HMR):', err.message);
  });
});

server.on('upgrade', (request, socket, head) => {
  try {
    const url = new URL(request.url || '', `http://${request.headers.host || 'localhost'}`);
    if (url.pathname === '/ws/live') {
      wss.handleUpgrade(request, socket, head, (ws) => {
        wss.emit('connection', ws, request);
      });
    } else {
      fallbackWss.handleUpgrade(request, socket, head, (ws) => {
        fallbackWss.emit('connection', ws, request);
      });
    }
  } catch (upgradeErr) {
    console.warn('[Server] WebSocket upgrade error:', upgradeErr);
    try {
      socket.destroy();
    } catch (_) {}
  }
});

wss.on('connection', async (clientWs: WebSocket, req: http.IncomingMessage) => {
  // Parse query parameters for topic, grade and adaptive context
  const url = new URL(req.url || '', `http://${req.headers.host}`);
  const topic = url.searchParams.get('topic') || 'the requested subject';
  const grade = url.searchParams.get('grade') || 'the student level';
  const studentId = url.searchParams.get('studentId') || null;
  const subjectId = url.searchParams.get('subjectId') || null;

  console.log(`[WebSocket] Live session started for Topic: "${topic}", Grade: "${grade}", Student: ${studentId || 'anonymous'}`);

  // Fetch learner context for personalised system prompt
  let learnerContext = '';
  if (studentId && subjectId) {
    try {
      const learner = getLearner(studentId);
      if (learner) {
        const subject = learner.subjects[subjectId];
        if (subject) {
          const conceptSummaries = Object.values(subject.conceptStates)
            .sort((a, b) => b.lastVisited - a.lastVisited).slice(0, 8)
            .map(cs => {
              const parts = [`  - ${cs.label}: mastery ${cs.masteryScore}/100 (${cs.masteryLevel})`];
              if (cs.confirmedMisconceptions.length)
                parts.push(`    CONFIRMED MISCONCEPTIONS: ${cs.confirmedMisconceptions.join('; ')}`);
              if (cs.effectiveStrategies.length)
                parts.push(`    EFFECTIVE STRATEGIES: ${cs.effectiveStrategies.join(', ')}`);
              if (cs.ineffectiveStrategies.length)
                parts.push(`    AVOID THESE: ${cs.ineffectiveStrategies.join(', ')}`);
              return parts.join('\n');
            }).join('\n');

          learnerContext = `
\n\n=== LEARNER PROFILE ===
Student Name: ${learner.name}
Sessions Completed: ${subject.sessionCount}
Total Study Time: ${subject.totalMinutes} minutes

CONCEPT MASTERY:
${conceptSummaries || '  No prior learning recorded — start from the beginning.'}

GLOBAL INSIGHTS: ${learner.globalInsights.slice(-2).join(' | ') || 'None yet.'}
=== END LEARNER PROFILE ===\n`;
        }
      }
    } catch (lcErr) {
      console.warn('[WebSocket] Could not load learner context:', lcErr);
    }
  }

  // ── Session-scoped teaching state ────────────────────────────
  // The voice model does not hold the learner model; the server does.
  // Every diagnostic decision is made here against persisted evidence.
  const liveCurriculum = subjectId
    ? (subjectId === 'pythagoras' ? PYTHAGORAS_CURRICULUM : getCurriculum(subjectId))
    : PYTHAGORAS_CURRICULUM;

  let liveConcept: CurriculumConcept | undefined = liveCurriculum?.concepts?.[0];
  let liveStrategy: TeachingStrategy = 'direct_explanation';
  let lastQuestionAsked = '';
  let lastQuestionAt = Date.now();

  if (studentId && subjectId && liveCurriculum) {
    try {
      ensureSubject(studentId, subjectId, liveCurriculum.label, liveCurriculum.grade, liveCurriculum.source);
      const learner = getLearner(studentId);
      const mastered = Object.values(learner?.subjects?.[subjectId]?.conceptStates || {})
        .filter(cs => cs.masteryScore >= MASTERY_THRESHOLD)
        .map(cs => cs.conceptId);
      const next = nextUnmasteredConcept(liveCurriculum, mastered) || liveCurriculum.concepts[0];
      if (next) {
        liveConcept = next;
        ensureConceptState(studentId, subjectId, next.id, next.label);
        const cs = getConceptState(studentId, subjectId, next.id);
        // Resume with a strategy that has not already failed this child.
        const failed = cs?.ineffectiveStrategies || [];
        if (failed.includes(liveStrategy)) liveStrategy = 'worked_example';
      }
    } catch (e) {
      console.warn('[WebSocket] could not resolve live concept:', e);
    }
  }

  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) {
    console.error('[WebSocket] GEMINI_API_KEY is missing');
    clientWs.send(
      JSON.stringify({
        type: 'error',
        code: 'API_KEY_MISSING',
        message: 'GEMINI_API_KEY is required for live voice interaction. Please configure it in AI Studio settings.',
      })
    );
    clientWs.close();
    return;
  }

  let liveSession: any = null;
  let isClosed = false;

  // ─────────────────────────────────────────────────────────────
  // Server-side tool execution.
  //
  // The voice model is deliberately NOT trusted to judge an answer.
  // It reports what the child said; the diagnostic pipeline decides what
  // that means and hands back a single imperative instruction. This is
  // what stops the tutor from stating the answer and moving on.
  // ─────────────────────────────────────────────────────────────
  // Guards the blocking window: only one diagnosis at a time per session.
  let assessInFlight = false;
  // A diagnosis that arrived after its turn, to be folded into the next one.
  let carriedInsight: string | null = null;

  type AnswerCtx = {
    questionAsked: string; childAnswer: string; childReasoning: string;
    promptType: 'teach' | 'check' | 'probe' | 'transfer'; latencyMs: number;
  };

  /** Records a real diagnosis as evidence and builds the payload for the model + UI. */
  function applyAssessment(assessment: ReasoningAssessment, concept: CurriculumConcept, ctx: AnswerCtx, late: boolean) {
    const candidates = assessment.candidateMisconceptionIds.map(id => ({
      id, text: misconceptionText(concept, id) || id,
    }));

    let record: ReturnType<typeof recordReasoningEvidence> = null;
    if (studentId && subjectId) {
      ensureConceptState(studentId, subjectId, concept.id, concept.label);
      record = recordReasoningEvidence({
        studentId, subjectId,
        conceptId: concept.id,
        difficultyLevel: concept.difficultyLevel,
        promptType: ctx.promptType,
        questionAsked: ctx.questionAsked,
        childAnswer: ctx.childAnswer,
        childReasoning: ctx.childReasoning,
        classification: assessment.classification,
        understandingDepth: assessment.understandingDepth,
        candidateMisconceptions: candidates,
        confidence: assessment.confidence,
        strategyInUse: liveStrategy,
        responseLatencyMs: ctx.latencyMs,
      });
    }

    let instruction = assessment.tutorGuidance;
    if (assessment.shouldProbe && assessment.probeQuestion) {
      instruction =
        `Ask EXACTLY this and then stop talking: "${assessment.probeQuestion}" ` +
        `Do not explain, do not hint, do not reveal the answer, do not say whether the ` +
        `previous answer was right. Wait for their reply, then call assess_child_reasoning ` +
        `again with promptType "probe".`;
    } else if (assessment.classification === 'clear_reasoning') {
      instruction = `${assessment.tutorGuidance} Praise the specific move they made, not the child. Then move to the next idea.`;
    }
    const newlyConfirmed = record?.newlyConfirmed || [];
    if (newlyConfirmed.length > 0) {
      instruction +=
        ` A misconception is now CONFIRMED: "${newlyConfirmed[0].text}". ` +
        `Do not simply state the rule. Show them a case where THEIR method gives the wrong ` +
        `answer, let them see it break, then re-teach using a ${assessment.nextStrategy.replace(/_/g, ' ')}.`;
    }

    liveStrategy = assessment.nextStrategy;
    if (!late) {
      lastQuestionAsked = assessment.probeQuestion || ctx.questionAsked;
      lastQuestionAt = Date.now();
    }

    return {
      result: 'assessed',
      instruction,
      classification: assessment.classification,
      understandingDepth: assessment.understandingDepth,
      confidence: assessment.confidence,
      reasoningSummary: assessment.reasoningSummary,
      faultyStep: assessment.faultyStep || null,
      shouldProbe: assessment.shouldProbe,
      probeQuestion: late ? null : (assessment.probeQuestion || null),
      nextStrategy: assessment.nextStrategy,
      concept: { id: concept.id, label: concept.label },
      mastery: record ? { before: record.masteryBefore, after: record.masteryAfter, derivation: record.derivation } : null,
      misconceptions: record?.ledger?.map(m => ({ id: m.id, text: m.text, status: m.status })) || [],
    };
  }

  function summariseForLater(a: ReasoningAssessment, concept: CurriculumConcept): string {
    if (a.classification === 'clear_reasoning') return 'their earlier method was sound';
    const mis = a.candidateMisconceptionIds.map(id => misconceptionText(concept, id)).filter(Boolean);
    if (mis.length) return `their earlier method suggests this misconception: "${mis[0]}"`;
    return 'their earlier method was unclear';
  }

  /** Prepends a late diagnosis to the next instruction, once. */
  function withCarriedInsight(payload: Record<string, any>): Record<string, any> {
    if (!carriedInsight || !payload.instruction) return payload;
    const note = `Background from checking their PREVIOUS answer: ${carriedInsight}. Use this, but do not mention it. `;
    carriedInsight = null;
    return { ...payload, instruction: note + payload.instruction };
  }

  async function handleLiveTool(name: string, args: Record<string, any>): Promise<Record<string, any>> {
    // ── Diagnostic path ──────────────────────────────────────
    if (name === 'assess_child_reasoning') {
      if (!liveConcept) {
        return {
          result: 'no_curriculum',
          instruction: 'Ask the child to explain how they worked it out before you say anything about the answer.',
        };
      }

      const questionAsked = String(args.questionAsked || lastQuestionAsked || '');
      const childAnswer = String(args.childAnswer || '');
      const childReasoning = String(args.childReasoning || '');
      const promptType = (['teach','check','probe','transfer'].includes(args.promptType)
        ? args.promptType : 'check') as 'teach' | 'check' | 'probe' | 'transfer';
      const latencyMs = Math.max(0, Date.now() - lastQuestionAt);

      // If the child has given an answer but no method yet, do not assess.
      // Elicit first. Assessing an unexplained answer is exactly the failure
      // mode we are engineered against.
      if (!childReasoning.trim()) {
        lastQuestionAsked = questionAsked;
        return {
          result: 'reasoning_required',
          classification: 'no_reasoning_given',
          instruction:
            'Do NOT say whether that is right or wrong, and do NOT give the answer. ' +
            'Ask the child how they worked it out, then wait for them to finish. ' +
            'Call assess_child_reasoning again once they have told you their method.',
        };
      }

      // Duplicate call while we are still checking (the child repeated
      // themselves and the model re-fired). Answer instantly; do not re-assess.
      if (assessInFlight) {
        return {
          result: 'already_checking',
          instruction:
            'You are already checking this answer. Do not ask the question again and do not repeat ' +
            'yourself. At most say "Got it" and wait.',
        };
      }

      const concept = liveConcept;
      const conceptState = studentId && subjectId
        ? getConceptState(studentId, subjectId, concept.id)
        : null;
      const ctx = { questionAsked, childAnswer, childReasoning, promptType, latencyMs };

      // Tell the browser straight away so the child can SEE they were heard.
      if (clientWs.readyState === WebSocket.OPEN) {
        clientWs.send(JSON.stringify({ type: 'tool_pending', name }));
      }

      assessInFlight = true;
      let result: DeadlineResult;
      try {
        result = await assessReasoningWithDeadline({
          concept,
          conceptState,
          questionAsked,
          childAnswer,
          childReasoning,
          expectedAnswer: args.expectedAnswer ? String(args.expectedAnswer) : undefined,
          currentStrategy: liveStrategy,
          apiKey: apiKey!,
        }, ASSESS_BUDGET_MS);
      } finally {
        assessInFlight = false;
      }
      console.log(`[assess] ${result.elapsedMs}ms ${result.timedOut ? 'DEADLINE — transfer move sent, diagnosis continues' : 'in budget'}`);

      if (!result.timedOut) {
        const payload = applyAssessment(result.assessment, concept, ctx, false);
        return withCarriedInsight(payload);
      }

      // Late path: the diagnosis lands after the conversation has moved on.
      // It is recorded as evidence and shown in the UI, and folded into the
      // tutor's NEXT instruction. The placeholder is never recorded as evidence.
      result.full.then(full => {
        if (isClosed) return;
        const payload = applyAssessment(full, concept, ctx, true);
        carriedInsight = summariseForLater(full, concept);
        if (clientWs.readyState === WebSocket.OPEN) {
          clientWs.send(JSON.stringify({ type: 'assessment_update', serverResult: payload }));
        }
      }).catch(e => console.warn('[assess] late diagnosis failed:', e?.message || e));

      lastQuestionAt = Date.now();
      return withCarriedInsight({
        result: 'checking',
        classification: 'pending',
        instruction: result.assessment.tutorGuidance,
        concept: { id: concept.id, label: concept.label },
      });
    }

    // ── Confusion signals ────────────────────────────────────
    if (name === 'record_confusion_signal') {
      const signal = String(args.signal || 'dont_understand');
      if (studentId && subjectId && liveConcept) {
        try {
          ensureConceptState(studentId, subjectId, liveConcept.id, liveConcept.label);
          addGlobalInsight(studentId, `Asked for help (${signal}) on ${liveConcept.label}${args.aboutWhat ? ': ' + args.aboutWhat : ''}`);
        } catch (_) {}
      }
      const responses: Record<string, string> = {
        dont_understand:      'Thank them plainly for saying so, then re-teach that ONE point using a completely different representation. Do not repeat the words you just used.',
        repeat_differently:   'Do not repeat yourself. Switch representation entirely — if you explained it, show it; if you showed it, use a concrete everyday example.',
        too_fast:             'Slow down. Take the last idea and break it into two smaller steps. Check after each one.',
        guessed:              'Treat admitting a guess as a good thing and say so. Then go back one step and rebuild from what they DO know.',
        silent:               'Do not fill the silence with the answer. Offer a smaller entry point: ask them what the first thing they would work out is.',
        disagrees_with_tutor: 'Take it seriously. Re-examine your own statement out loud. If they are right, say so clearly and thank them.',
      };
      return {
        result: 'recorded',
        instruction: (responses[signal] || responses.dont_understand) +
          ' Never sigh, never say "as I said", never imply this should have been obvious.',
      };
    }

    // ── Board tool: the child's thinking goes up on the board ─
    if (name === 'show_student_thinking') {
      return {
        result: 'displayed',
        instruction: "The child's method is now on the board beside the figure. Refer to it as their method, not as a mistake.",
      };
    }

    // ── Board tools: executed in the browser ─────────────────
    if (name === 'pose_quiz' && args.question) {
      lastQuestionAsked = String(args.question);
      lastQuestionAt = Date.now();
    }
    return { result: 'ok' };
  }


  // Which voice configuration this session uses (see src/live/liveConfig.ts).
  const voiceMode: VoiceMode = process.env.VOICE_MODE === 'adaptive' ? 'adaptive' : 'classic';
  const dynamicSystemInstruction = voiceMode === 'adaptive'
    ? adaptiveSystemInstruction(topic, grade, learnerContext)
    : classicSystemInstruction(topic, grade, learnerContext);


  // ─────────────────────────────────────────────────────────────
  // Live connection management.
  //
  // Google closes a Live connection after ~10 minutes and an audio session
  // after 15 unless context compression is on. It warns first with goAway.
  // Previously none of that was handled and the browser was never told, so
  // a dropped session looked like a hung tutor. Now:
  //   - context compression is on (sessions are not capped at 15 minutes)
  //   - a resumption handle is kept, and on goAway or an unexpected close the
  //     server reconnects to Gemini behind the scenes; the child's browser
  //     connection stays open throughout
  //   - tool calls Gemini cancels (child barged in) are never answered
  //   - every event is written to logs/voice-*.log for diagnosis
  // ─────────────────────────────────────────────────────────────
  const LIVE_MODEL = process.env.LIVE_MODEL || 'gemini-3.8-live';
  const vlog = openVoiceLog(topic);
  let resumeHandle: string | null = null;
  let generation = 0;
  let reconnecting = false;
  const cancelledCalls = new Set<string>();
  let lastUserSpeechAt = 0;
  let awaitingReplyAfterUser = false;

  const toClient = (msg: Record<string, any>) => {
    if (clientWs.readyState === WebSocket.OPEN) clientWs.send(JSON.stringify(msg));
  };

  const ai = new GoogleGenAI({ apiKey, httpOptions: { headers: { 'User-Agent': 'aistudio-build' } } });

  function handleServerMessage(message: LiveServerMessage) {
    if (isClosed || clientWs.readyState !== WebSocket.OPEN) return;
    const sc: any = message.serverContent;

    // Audio + text from the model
    const modelParts = message.serverContent?.modelTurn?.parts;
    if (modelParts && Array.isArray(modelParts)) {
      for (const part of modelParts) {
        if (part.inlineData?.data) {
          if (awaitingReplyAfterUser) {
            awaitingReplyAfterUser = false;
            vlog('reply_latency', { ms: Date.now() - lastUserSpeechAt });
          }
          toClient({ type: 'audio_out', data: part.inlineData.data });
        }
        if (part.text) toClient({ type: 'output_transcript', text: part.text });
      }
    }

    const outTrans = sc?.outputTranscription?.text ?? sc?.outputAudioTranscription?.text;
    if (outTrans) { toClient({ type: 'output_transcript', text: outTrans }); vlog('tutor_said', { text: outTrans }); }
    const inTrans = sc?.inputTranscription?.text ?? sc?.inputAudioTranscription?.text;
    if (inTrans) {
      lastUserSpeechAt = Date.now();
      awaitingReplyAfterUser = true;
      toClient({ type: 'input_transcript', text: inTrans });
      vlog('child_said', { text: inTrans });
    }
    if (sc?.interrupted) { toClient({ type: 'interrupted' }); vlog('interrupted'); }
    if (sc?.turnComplete) { toClient({ type: 'turn_complete' }); vlog('turn_complete'); }

    // Session housekeeping
    if (message.sessionResumptionUpdate?.resumable && message.sessionResumptionUpdate.newHandle) {
      resumeHandle = message.sessionResumptionUpdate.newHandle;
    }
    if (message.goAway) {
      vlog('go_away', { timeLeft: message.goAway.timeLeft });
      // Resuming needs a handle; without one a reconnect would be a blank session.
      if (voiceMode === 'adaptive' && resumeHandle) void resume('goAway');
    }
    if (message.toolCallCancellation?.ids?.length) {
      for (const id of message.toolCallCancellation.ids) cancelledCalls.add(id);
      vlog('tool_cancelled', { ids: message.toolCallCancellation.ids });
      toClient({ type: 'tool_cancelled', ids: message.toolCallCancellation.ids });
    }

    // Tool calls
    const functionCalls = message.toolCall?.functionCalls;
    if (functionCalls && Array.isArray(functionCalls) && functionCalls.length > 0) {
      if (voiceMode === 'classic') {
        // Original behaviour: mirror to the browser, answer "ok" at once.
        for (const call of functionCalls) {
          vlog('tool_call', { name: call.name, args: call.args });
          toClient({ type: 'tool_call', id: call.id, name: call.name, args: call.args || {} });
          try {
            liveSession.sendToolResponse({ functionResponses: [{ id: call.id, name: call.name, response: { result: 'ok' } }] });
          } catch (respErr: any) {
            vlog('tool_response_failed', { name: call.name, error: String(respErr?.message || respErr) });
          }
        }
        return;
      }
      const sessionForCalls = liveSession;   // answer on the connection that asked
      for (const call of functionCalls) {
        const t0 = Date.now();
        vlog('tool_call', { name: call.name, args: call.args });
        void (async () => {
          let payload: Record<string, any> = { result: 'ok' };
          try {
            payload = await handleLiveTool(call.name || '', (call.args || {}) as Record<string, any>);
          } catch (toolErr: any) {
            console.error(`[Tool Call] ${call.name} failed:`, toolErr?.message || toolErr);
            payload = {
              result: 'error',
              instruction: 'The assessor is unavailable. Do NOT give the child the answer. Ask them to walk you through their working, starting with the first step.',
            };
          }
          toClient({ type: 'tool_call', id: call.id, name: call.name, args: call.args || {}, serverResult: payload });

          // A cancelled call must not be answered, and a call from a connection
          // we have since replaced cannot be answered on the new one.
          if (call.id && cancelledCalls.has(call.id)) { vlog('tool_response_skipped', { name: call.name, reason: 'cancelled' }); return; }
          if (isClosed || liveSession !== sessionForCalls) { vlog('tool_response_skipped', { name: call.name, reason: 'stale connection' }); return; }
          try {
            liveSession.sendToolResponse({ functionResponses: [{ id: call.id, name: call.name, response: payload }] });
            vlog('tool_response', { name: call.name, ms: Date.now() - t0, result: payload.result });
          } catch (respErr: any) {
            vlog('tool_response_failed', { name: call.name, error: String(respErr?.message || respErr) });
          }
        })();
      }
    }
  }

  async function connectLive(handle: string | null) {
    const myGen = ++generation;
    const session = await ai.live.connect({
      model: LIVE_MODEL,
      config: {
        responseModalities: [Modality.AUDIO],
        ...liveConfigFor(voiceMode, dynamicSystemInstruction, handle),
      } as any,
      callbacks: {
        onopen: () => vlog('gemini_open', { generation: myGen, resumed: !!handle }),
        onmessage: (m: LiveServerMessage) => { if (myGen === generation) handleServerMessage(m); },
        onerror: (e: any) => {
          vlog('gemini_error', { generation: myGen, message: e?.message || String(e) });
          if (myGen === generation) toClient({ type: 'error', code: 'LIVE_ERROR', message: e?.message || 'Gemini Live error' });
        },
        onclose: (e: any) => {
          vlog('gemini_close', { generation: myGen, code: e?.code, reason: e?.reason || '' });
          // Replaced connections and intentional shutdowns are expected.
          if (myGen !== generation || isClosed) return;
          if (voiceMode === 'adaptive' && resumeHandle && !reconnecting) void resume(`closed ${e?.code ?? ''} ${e?.reason ?? ''}`.trim());
          else toClient({ type: 'session_closed', reason: e?.reason || `code ${e?.code ?? 'unknown'}` });
        },
      },
    });
    return session;
  }

  async function resume(why: string) {
    if (reconnecting || isClosed) return;
    reconnecting = true;
    vlog('resuming', { why, haveHandle: !!resumeHandle });
    toClient({ type: 'reconnecting' });
    const old = liveSession;
    try {
      liveSession = await connectLive(resumeHandle);
      try { old?.close(); } catch (_) {}
      vlog('resumed');
      toClient({ type: 'reconnected' });
    } catch (e: any) {
      vlog('resume_failed', { error: String(e?.message || e) });
      toClient({ type: 'session_closed', reason: 'could not reconnect' });
    } finally {
      reconnecting = false;
    }
  }

  try {
    liveSession = await connectLive(null);

    if (clientWs.readyState === WebSocket.OPEN) {
      clientWs.send(JSON.stringify({ type: 'session_ready' }));
      vlog('session_ready', { model: LIVE_MODEL, voiceMode, studentId, subjectId });

      // Kick off the lesson (first connection only — never on a resume).
      try {
        liveSession.sendClientContent({
          turns: [{ role: 'user', parts: [{ text: voiceMode === 'adaptive' ? adaptiveKickoff(topic, grade) : classicKickoff(topic, grade) }] }],
          turnComplete: true,
        });
      } catch (initErr) {
        console.error('[Gemini Live] Error sending initial turn:', initErr);
      }
    }
  } catch (err: any) {
    vlog('connect_failed', { error: String(err?.message || err) });
    console.error('[WebSocket] Failed to initialize Gemini Live session:', err);
    clientWs.send(
      JSON.stringify({
        type: 'error',
        code: 'INITIALIZATION_FAILED',
        message: err?.message || 'Failed to start Live session',
      })
    );
    clientWs.close();
    return;
  }

  // Mic level, summarised every 2 s into the session log. If the tutor
  // "waits" while the mic shows steady sound, background noise is keeping
  // Gemini's end-of-speech detection from firing.
  let micSum = 0, micPeak = 0, micN = 0, micLoud = 0, micLastLog = Date.now();
  const micStats = (b64: string) => {
    try {
      const buf = Buffer.from(b64, 'base64');
      const n = Math.floor(buf.length / 2);
      let sq = 0;
      for (let i = 0; i < n; i++) { const v = buf.readInt16LE(i * 2) / 32768; sq += v * v; }
      const rms = Math.sqrt(sq / Math.max(1, n));
      micSum += rms; micN++; micPeak = Math.max(micPeak, rms);
      if (rms > 0.02) micLoud++;           // roughly -34 dBFS: louder than a quiet room
      if (Date.now() - micLastLog >= 2000 && micN) {
        const db = (x: number) => Math.round(20 * Math.log10(Math.max(x, 1e-5)));
        vlog('mic', { avgDb: db(micSum / micN), peakDb: db(micPeak), loudPct: Math.round(micLoud / micN * 100) });
        micSum = micPeak = micN = micLoud = 0; micLastLog = Date.now();
      }
    } catch (_) {}
  };

  // Handle messages from client
  clientWs.on('message', (rawData) => {
    if (isClosed || !liveSession || reconnecting) return;

    try {
      const msg = JSON.parse(rawData.toString());

      if ((msg.type === 'audio' || msg.type === 'audio_in') && msg.data) {
        micStats(msg.data);
        liveSession.sendRealtimeInput({
          audio: {
            data: msg.data,
            mimeType: 'audio/pcm;rate=16000',
          },
        });
      } else if (msg.type === 'text' && msg.text) {
        liveSession.sendClientContent({
          turns: [{ role: 'user', parts: [{ text: msg.text }] }],
          turnComplete: true,
        });
      } else if (msg.type === 'tool_response') {
        console.log('[WebSocket] Tool execution confirmed:', msg.id);
      } else if (msg.type === 'close') {
        isClosed = true;
        clientWs.close();
      }
    } catch (parseErr) {
      console.error('[WebSocket] Failed to parse client message:', parseErr);
    }
  });

  clientWs.on('close', () => {
    isClosed = true;
    vlog('client_disconnected');
    console.log('[WebSocket] Client disconnected');
    if (liveSession) {
      try {
        liveSession.close();
      } catch (closeErr) {
        console.error('[Gemini Live] Error closing session:', closeErr);
      }
    }
  });
});

async function startServer() {
  // Vite middleware in dev or static files in production
  if (process.env.NODE_ENV !== 'production') {
    const { createServer: createViteServer } = await import('vite');
    const isHmrDisabled = process.env.DISABLE_HMR === 'true';
    const vite = await createViteServer({
      server: {
        middlewareMode: true,
        hmr: isHmrDisabled ? false : { server },
      },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    const distPath = path.join(process.cwd(), 'dist');
    app.use(express.static(distPath));
    app.get('*', (req, res) => {
      res.sendFile(path.join(distPath, 'index.html'));
    });
  }

  server.listen(PORT, '0.0.0.0', () => {
    console.log(`[Server] AI Tutor backend running on http://localhost:${PORT}`);
  });
}

startServer();
